httpx                       # async HTTP (LLM calls, git-less fetches) (§4)

python-dotenv               # load ANTHROPIC_API_KEY / OPENAI_API_KEY from .env

# ── Ingest / VCS ───────────────────────────────────────────────────────────
gitpython                   # clone git URLs → RepoModel (§9.1)

# ── Parsing (§12) ──────────────────────────────────────────────────────────
# NOTE: use tree-sitter-language-pack (bundles a correctly-matched Solidity
# grammar via PyCapsule). The standalone `tree-sitter-solidity` wheel returns a

# Import with:  from tree_sitter_language_pack import get_parser
tree-sitter                 # error-tolerant parser core — PRIMARY structure source
tree-sitter-language-pack   # bundled Solidity grammar + matched binding

# ── Graphs (§13) ───────────────────────────────────────────────────────────
networkx                    # the four graphs; adequate to ~5k nodes — do NOT use a graph DB

# ── LLM providers (§4) — imported ONLY under engine/llm/ ───────────────────
anthropic                   # Claude API — PRIMARY
openai                      # OpenAI — fallback / second opinion

# ── Static analysis (§11) — adapters, not dependencies ─────────────────────
slither-analyzer            # Slither (primary). Requires a working solc/compilation.
solc-select                 # manage solc versions for the RAW/direct-solc build path (§10.1)

# ── CLI / logging ──────────────────────────────────────────────────────────
rich                        # structured, readable CLI logs (§4)
typer                       # CLI entrypoint (the engine must run without the frontend, §4)
